# Required scope: rite. Parameters: TITLE (landed title), CULTURE (culture scope).
# Only vacant, active clerical regions are installed; existing holders are preserved.
# The faith and employer come from this rite, never from ROOT or a fixed patriarchate.
nex_create_clerical_holder_effect = {
	if = {
		limit = {
			exists = faith.religious_head
			exists = $TITLE$
			exists = $CULTURE$
			$TITLE$ = {
				has_clerical_region = yes
				NOT = { exists = holder }
				exists = title_preferred_capital_county
			}
		}
		save_scope_as = nex_clerical_holder_rite
		faith = {
			save_scope_as = nex_clerical_holder_faith
			religious_head = { save_scope_as = nex_clerical_holder_employer }
		}
		create_character = {
			template = simple_priest_character_template
			gender = male
			faith = scope:nex_clerical_holder_faith
			rite = scope:nex_clerical_holder_rite
			culture = $CULTURE$
			employer = scope:nex_clerical_holder_employer
			save_scope_as = nex_clerical_holder
		}
		create_title_and_vassal_change = {
			type = appointment
			save_scope_as = nex_clerical_holder_change
			add_claim_on_loss = no
		}
		$TITLE$ = {
			change_title_holder = {
				holder = scope:nex_clerical_holder
				change = scope:nex_clerical_holder_change
				government_base = scope:nex_clerical_holder_employer
			}
		}
		resolve_title_and_vassal_change = scope:nex_clerical_holder_change
		scope:nex_clerical_holder = {
			change_government = ecclesiastical_government
			situation:the_christian_church ?= { add_manual_participant = prev }
		}
	}
}
