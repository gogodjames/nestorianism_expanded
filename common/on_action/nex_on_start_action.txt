
on_game_start = {
    on_actions = { nex_on_game_start }
}

nex_on_game_start = {
    effect = {
        # Patriarch Flavour
        title:d_patriarchate_in_the_east = { set_coa = nex_patriarch_coa }

        # Malayali culture
        title:c_alupa = { set_county_culture = culture:nex_malayali }
        title:c_eranad = { set_county_culture = culture:nex_malayali }
        title:c_kerala = { set_county_culture = culture:nex_malayali }
        title:c_venadu = { set_county_culture = culture:nex_malayali }

        # Start with the shrine of St. Thomas built
        title:b_uttaramerur.title_province = {
            if = {
                limit = {
                    has_special_building = no
                    NOT = { has_building_or_higher = nex_saint_thomas_mount_01 }
                }
                add_special_building = nex_saint_thomas_mount_01
            }
        }

        # Convert the mainline Nestorian counties of Mongolic and Turkic cultures.
        rite:nestorian = {
            every_rite_county = {
                # Steppe Christians
                if = { 
                    limit = {
                        culture ?= {
                            OR = {
                                has_cultural_pillar = heritage_mongolic
                                has_cultural_pillar = heritage_turkic
                            }
                        }
                    }
                    set_county_rite = rite:zex_arkagun_rite
                }
                # Socotra
                else_if = {
                    limit = { title:c_socotra ?= this }
                    set_county_rite = rite:zex_socotran_rite
                }
                else = {} # Stay Mainline
            }
        }

        # Generate Holders for the ones that don't have historical ones at the time
        give_interior_provinces_holders_effect = yes # Mesopotamia/Persia
        give_exterior_provinces_holders_effect = yes # Elsewhere

        # Refresh Head of Rites
        faith:nestorian = {
            every_faith_rite = { force_refresh_head_of_rite = yes }
        }
        
        # Historical Church of the East monasteries.
        nex_initialize_historical_monasteries_effect = yes
    }
}
