
on_game_start = {
    on_actions = { nex_on_game_start }
}

nex_on_game_start = {
    
    effect = {
        # Patriarch Flavour
        title:d_patriarchate_in_the_east = { set_coa = nex_patriarch_coa }

        # Malayali culture
        title:c_alupa = { set_county_culture = culture:nex_malayali }
        title:c_eranad = { set_county_culture = culture:nex_malayali }
        title:c_kerala = { set_county_culture = culture:nex_malayali }
        title:c_venadu = { set_county_culture = culture:nex_malayali }

        # 867, Jingjiao was recently persecuted by the Tang, so create some refugees for them
        if = {
            limit = { game_start_date = 867.1.1 }

            rite:zex_jingjiao_rite = {
                set_variable = {
                    name = associated_county
                    value = title:c_jingzhao
                }
                set_variable = {
                    name = associated_culture
                    value = culture:han
                }
                set_variable = {
                    name = converted_development
                    value = 40
                }
            }
        }

        # Start with the shrine of St. Thomas built
        title:b_kanchipuram.title_province = {
            if = {
                limit = {
                    has_special_building = yes
                    NOT = { has_building_or_higher = nex_saint_thomas_mount_01 }
                }
                remove_building = generic_university
                add_special_building = nex_saint_thomas_mount_01
            }
        }

        # Enshrine Thomas's relic at St. Thomas Mount
        if = {
            limit = {
                exists = holy_site:nex_saint_thomas_mount
                exists = faith:nestorian.religious_head
                holy_site:nex_saint_thomas_mount = { holy_site_artifact_count < 3 }
            }
            holy_site:nex_saint_thomas_mount = { save_scope_as = relic_site }
            faith:nestorian = {
                save_scope_as = relic_faith
                religious_head = { save_scope_as = relic_owner }
            }
            clear_saved_scope = newly_created_artifact
            set_artifact_rarity_illustrious = yes
            scope:relic_owner = {
                create_artifact = {
                    name = artifact_relic_thomas_name
                    description = artifact_relic_thomas_desc
                    type = pedestal
                    template = christian_artifact_template
                    visuals = pedestal_christian_relic
                    wealth = scope:wealth
                    quality = scope:quality
                    history = { type = created_before_history }
                    modifier = artifact_placeholder_modifier
                    save_scope_as = newly_created_artifact
                }
            }
            finalize_starting_unique_relic_effect = yes
            if = {
                limit = {
                    exists = scope:newly_created_artifact
                    scope:relic_site = {
                        any_holy_site_enshrined_artifact = {
                            this = scope:newly_created_artifact
                        }
                    }
                }
            }
        }

        # Enshrine Kollam Copper Plates at Kerala Holy Site
        if = {
            limit = {
                exists = holy_site:kerala
                exists = faith:nestorian.religious_head
                holy_site:kerala = { holy_site_artifact_count < 3 }
            }
            holy_site:kerala = { save_scope_as = relic_site }
            faith:nestorian = {
                save_scope_as = relic_faith
                religious_head = { save_scope_as = relic_owner }
            }
            clear_saved_scope = newly_created_artifact
            set_artifact_rarity_masterwork = yes
            scope:relic_owner = {
                create_artifact = {
                    name = artifact_relic_kerala_name
                    description = artifact_relic_kerala_desc
                    type = pedestal
                    template = christian_artifact_template
                    visuals = pedestal_christian_relic
                    wealth = scope:wealth
                    quality = scope:quality
                    history = { type = created_before_history}
                    modifier = artifact_placeholder_modifier
                    save_scope_as = newly_created_artifact
                }
            }
            finalize_starting_unique_relic_effect = yes
            if = {
                limit = {
                    exists = scope:newly_created_artifact
                    scope:relic_site = {
                        any_holy_site_enshrined_artifact = {
                            this = scope:newly_created_artifact
                        }
                    }
                }
            }
        }

        # Enshrine Nestorian Stele at Chang'an Holy Site
        if = {
            limit = {
                exists = holy_site:chang_an
                exists = faith:nestorian.religious_head
                holy_site:chang_an = { holy_site_artifact_count < 3 }
            }
            holy_site:chang_an = { save_scope_as = relic_site }
            faith:nestorian = {
                save_scope_as = relic_faith
                religious_head = { save_scope_as = relic_owner }
            }
            clear_saved_scope = newly_created_artifact
            set_artifact_rarity_masterwork = yes
            scope:relic_owner = {
                create_artifact = {
                    name = artifact_relic_nestorian_stele_name
                    description = artifact_relic_nestorian_stele_desc
                    type = pedestal
                    template = christian_artifact_template
                    visuals = pedestal_christian_relic
                    wealth = scope:wealth
                    quality = scope:quality
                    history = {type = created_before_history }
                    modifier = artifact_placeholder_modifier
                    save_scope_as = newly_created_artifact
                }
            }
            finalize_starting_unique_relic_effect = yes
            if = {
                limit = {
                    exists = scope:newly_created_artifact
                    scope:relic_site = {
                        any_holy_site_enshrined_artifact = {
                            this = scope:newly_created_artifact
                        }
                    }
                }
            }
        }

        # Convert the mainline Nestorian counties of Mongolic and Turkic cultures.
        rite:nestorian = {
            every_rite_county = {
                # Steppe Christians
                if = { 
                    limit = {
                        culture ?= {
                            OR = {
                                has_cultural_pillar = heritage_mongolic
                                has_cultural_pillar = heritage_turkic
                            }
                        }
                    }
                    set_county_rite = rite:zex_arkagun_rite
                }
                # Socotra
                else_if = {
                    limit = { title:c_socotra ?= this }
                    set_county_rite = rite:zex_socotran_rite
                }
                else = {} # Stay Mainline
            }
        }

        # Generate Holders for the ones that don't have historical ones at the time
        give_interior_provinces_holders_effect = yes # Mesopotamia/Persia
        give_exterior_provinces_holders_effect = yes # Elsewhere

        # Favor Abdisho II, metropolitan of Nisibis, as the next patriarch in 1066.
        if = {
            limit = {
                game_start_date = 1066.9.15
                exists = character:patriarch_in_the_east_70
            }
            title:d_patriarchate_in_the_east ?= {
                change_appointment_investment = {
                    target = character:patriarch_in_the_east_70
                    value = 500
                }
            }
        }

        # Refresh Head of Rites
        faith:nestorian = {
            every_faith_rite = { force_refresh_head_of_rite = yes }
        }
        
        # Historical Church of the East monasteries.
        nex_initialize_historical_monasteries_effect = yes
    }
}
